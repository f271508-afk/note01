Google Sheets API 設定

請輸入您的 Google Apps Script 部署網址 (Web App URL)。若未設定，系統將僅使用瀏覽器本機儲存（LocalStorage），設定後可雙向同步至 Google 試算表。

為什麼修改後一定要重新部署？以及 Google Apps Script 注意事項：

  1. Google 網頁應用程式的快取機制：當您在 Apps Script 程式碼中修改邏輯（例如修正 `update` 動作的欄位寫入）時，若不重新建立新版本部署，Google 仍會執行舊版的程式碼，因此雲端試算表便不會接收到更新。
  2. 如何正確重新部署：
    • 進入您的 Google Apps Script 專案。
    • 點擊右上角藍色按鈕 「部署」 > 「管理部署」 (Manage Deployments)。
    • 點擊左側的齒輪圖示，或點擊右上角 「新增部署」 (New Deployment)。
    • 類型選擇 「網頁應用程式」 (Web app)。
    • 版本 (Version) 請務必下拉選擇 「新版本」 (New version)——這是最關鍵的步驟！
    • 執行身分維持「我」，誰可以存取 (Who has access) 務必選擇 「任何人」 (Anyone)。
    • 點擊「部署」並授權，取得新的網頁應用程式網址（或維持舊網址若用管理部署覆蓋版號）。

Google Apps Script 完整程式碼範本

請將以下完整程式碼清空並貼入您的 Google Apps Script 編輯器中，務必重新部署為「新的版本」且存取權限設為「任何人」：

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(15000);
  try {
    var rawData = e.postData.contents;
    var data = JSON.parse(rawData);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    
    var action = String(data.action || "").trim();
    var targetId = String(data.id || "").trim();
    
    Logger.log("Action received: " + action + ", Target ID: " + targetId);
    
    if (action === 'save') {
      sheet.appendRow([
        targetId, 
        String(data.title || ""), 
        String(data.content || ""), 
        String(data.imageUrl || ""), 
        String(data.timestamp || ""), 
        String(data.category || "其他"), 
        data.pinned ? "true" : "false"
      ]);
      return ContentService.createTextOutput(JSON.stringify({status: 'success', action: 'save'})).setMimeType(ContentService.MimeType.JSON);
    } 
    else if (action === 'update') {
      var found = false;
      for (var i = 1; i < rows.length; i++) {
        var rowId = String(rows[i][0]).trim();
        if (rowId === targetId) {
          sheet.getRange(i + 1, 2, 1, 6).setValues([[
            String(data.title || ""), 
            String(data.content || ""), 
            String(data.imageUrl || ""), 
            String(data.timestamp || ""), 
            String(data.category || "其他"), 
            data.pinned ? "true" : "false"
          ]]);
          found = true;
          break;
        }
      }
      
      if (!found) {
        sheet.appendRow([
          targetId, 
          String(data.title || ""), 
          String(data.content || ""), 
          String(data.imageUrl || ""), 
          String(data.timestamp || ""), 
          String(data.category || "其他"), 
          data.pinned ? "true" : "false"
        ]);
      }
      
      return ContentService.createTextOutput(JSON.stringify({status: 'success', action: 'update', found: found})).setMimeType(ContentService.MimeType.JSON);
    } 
    else if (action === 'delete') {
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][0]).trim() === targetId) {
          sheet.deleteRow(i + 1);
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({status: 'success', action: 'delete'})).setMimeType(ContentService.MimeType.JSON);
    } 
    else if (action === 'updatePin') {
      for (var i = 1; i < rows.length; i++) {
        if (String(rows[i][0]).trim() === targetId) {
          sheet.getRange(i + 1, 7).setValue(data.pinned ? "true" : "false");
          break;
        }
      }
      return ContentService.createTextOutput(JSON.stringify({status: 'success', action: 'updatePin'})).setMimeType(ContentService.MimeType.JSON);
    }
    
    return ContentService.createTextOutput(JSON.stringify({status: 'invalid_action', received: action})).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({status: 'error', message: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function doGet() {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var rows = sheet.getDataRange().getValues();
  var notes = [];
  for (var i = 1; i < rows.length; i++) {
    if (!rows[i][0]) continue;
    notes.push({
      id: String(rows[i][0]).trim(),
      title: String(rows[i][1] || ''),
      content: String(rows[i][2] || ''),
      imageUrl: String(rows[i][3] || ''),
      timestamp: String(rows[i][4] || ''),
      category: String(rows[i][5] || '其他'),
      pinned: String(rows[i][6]) === 'true' || rows[i][6] === true
    });
  }
  var output = ContentService.createTextOutput(JSON.stringify(notes));
  output.setMimeType(ContentService.MimeType.JSON);
  return output;
}
